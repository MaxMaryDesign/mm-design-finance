function doPost(e){
  try{
    var data=JSON.parse(e.postData?e.postData.contents:e.parameter.d);
    saveData(data);
    return ContentService.createTextOutput(JSON.stringify({status:'ok'}))
      .setMimeType(ContentService.MimeType.JSON);
  }catch(err){
    return ContentService.createTextOutput(JSON.stringify({status:'error',msg:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e){
  try{
    var cb=e.parameter.callback||'';
    if(e.parameter.d){
      var data=JSON.parse(decodeURIComponent(e.parameter.d));
      saveData(data);
      var res=JSON.stringify({status:'ok'});
      return ContentService.createTextOutput(cb?cb+'('+res+')':res)
        .setMimeType(cb?ContentService.MimeType.JAVASCRIPT:ContentService.MimeType.JSON);
    }
    // Load raw JSON from data sheet
    var sh=SpreadsheetApp.getActiveSpreadsheet().getSheetByName('data');
    if(!sh||!sh.getRange(1,1).getValue()){
      var res=JSON.stringify({status:'empty'});
      return ContentService.createTextOutput(cb?cb+'('+res+')':res)
        .setMimeType(cb?ContentService.MimeType.JAVASCRIPT:ContentService.MimeType.JSON);
    }
    var val=sh.getRange(1,1).getValue();
    var res=JSON.stringify({status:'ok',data:JSON.parse(val)});
    return ContentService.createTextOutput(cb?cb+'('+res+')':res)
      .setMimeType(cb?ContentService.MimeType.JAVASCRIPT:ContentService.MimeType.JSON);
  }catch(err){
    var cb=e.parameter?e.parameter.callback||'':'';
    var res=JSON.stringify({status:'error',msg:err.toString()});
    return ContentService.createTextOutput(cb?cb+'('+res+')':res)
      .setMimeType(cb?ContentService.MimeType.JAVASCRIPT:ContentService.MimeType.JSON);
  }
}

function saveData(data){
  var ss=SpreadsheetApp.getActiveSpreadsheet();

  // ── Sheet: data (raw JSON backup) ──
  var sh=ss.getSheetByName('data')||ss.insertSheet('data');
  sh.clearContents();
  sh.getRange(1,1).setValue(JSON.stringify(data));

  // ── Sheet: Projects ──
  var ps=ss.getSheetByName('Projects')||ss.insertSheet('Projects');
  ps.clearContents();
  ps.getRange(1,1,1,9).setValues([['Name','Client','Status','Area m²','Budget (AED)','Start','Deadline','Total Income','Total Expenses']]);
  ps.getRange(1,1,1,9).setFontWeight('bold').setBackground('#697C70').setFontColor('#F2EFE2');
  var projs=data.projects||[];
  projs.forEach(function(p,i){
    var inc=(p.income||[]).reduce(function(s,r){return s+r.amount;},0);
    var exp=(p.expenses||[]).filter(function(r){return r.paid!==false;}).reduce(function(s,r){return s+r.amount;},0);
    ps.getRange(i+2,1,1,9).setValues([[
      p.name||'',p.client||'',p.status||'',p.area||0,
      (p.services||[]).filter(function(s){return s.on;}).reduce(function(sum,s){return sum+s.pricePerSqm*(p.area||0);},0),
      p.start||'',p.deadline||'',inc,exp
    ]]);
  });
  ps.autoResizeColumns(1,9);

  // ── Sheet: Income ──
  var is=ss.getSheetByName('Income')||ss.insertSheet('Income');
  is.clearContents();
  is.getRange(1,1,1,4).setValues([['Date','Project','Description','Amount (AED)']]);
  is.getRange(1,1,1,4).setFontWeight('bold').setBackground('#4CAF79').setFontColor('#fff');
  var row=2;
  projs.forEach(function(p){
    (p.income||[]).forEach(function(r){
      is.getRange(row++,1,1,4).setValues([[r.date||'',p.name||'',r.desc||'',r.amount||0]]);
    });
  });
  if(row>2) is.autoResizeColumns(1,4);

  // ── Sheet: Expenses ──
  var es=ss.getSheetByName('Expenses')||ss.insertSheet('Expenses');
  es.clearContents();
  es.getRange(1,1,1,5).setValues([['Date','Project','Category','Description','Amount (AED)']]);
  es.getRange(1,1,1,5).setFontWeight('bold').setBackground('#B85C5C').setFontColor('#fff');
  var row2=2;
  projs.forEach(function(p){
    (p.expenses||[]).filter(function(r){return r.paid!==false;}).forEach(function(r){
      es.getRange(row2++,1,1,5).setValues([[r.date||'',p.name||'',r.cat||'',r.desc||'',r.amount||0]]);
    });
  });
  if(row2>2) es.autoResizeColumns(1,5);

  // ── Sheet: Team ──
  var ts=ss.getSheetByName('Team')||ss.insertSheet('Team');
  ts.clearContents();
  ts.getRange(1,1,1,6).setValues([['Name','Role','Location','Type','Salary (AED/mo)','Start Date']]);
  ts.getRange(1,1,1,6).setFontWeight('bold').setBackground('#697C70').setFontColor('#F2EFE2');
  (data.team||[]).forEach(function(e,i){
    ts.getRange(i+2,1,1,6).setValues([[e.name||'',e.role||'',e.loc||'',e.type||'staff',e.salary||0,e.start||'']]);
  });
  ts.autoResizeColumns(1,6);
}
